# Реализуем робастную проверку на выбросы через медианный центр (median) и 
# среднее абсолютное отклонение от медианы (MAD). Используем стандартный Pandas и NumPy

import pandas as pd
import numpy as np

# Загружаем файл
df = pd.read_csv('data.csv')

# Проверим структуру данных
print(f'Размерность таблицы: {df.shape}')

# Отбираем столбцы 'date' и 'tat_min'
df_check = df[['date', 'tat_min']]

# Рассчитываем медиану срока рассмотрения заявки
med_tat = df_check['tat_min'].median()

# Вычисляем MAD (среднее абсолютное отклонение от медианы)
mad_tat = np.median(np.abs(df_check['tat_min'] - med_tat))

# Определяем порог выбросов: выбрасываем наблюдения, где разница больше трёх стандартных отклонений
threshold = 3 * mad_tat
outlier_mask = np.abs(df_check['tat_min'] - med_tat) > threshold

# Создаём новую колонку с меткой выбросов
df_check['is_outlier'] = outlier_mask

# Посмотрим количество выбросов и сами выбросы
n_outliers = sum(outlier_mask)
print(f'Общее число выбросов: {n_outliers}')

# Сформируем строку для отчёта
report_string = f'''Робастный анализ выбросов по срокам обработки заявок:
Дата начала анализа: дата первой записи
Дата окончания анализа: дата последней записи
Число записей до фильтрации: {len(df)}
Число записей после фильтрации: {len(df_check)}
Число выбросов: {n_outliers}
'''

# Выводим отчет и найденные выбросы
print(report_string)
print(df_check[df_check['is_outlier']])

# Печатаем финальный результат в удобочитаемом виде
print(df_check.head(1)) # выводим первую строку результата
