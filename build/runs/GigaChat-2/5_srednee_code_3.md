```python
import pandas as pd
import numpy as np

# Чтение исходных данных
data = pd.read_csv('your_data.csv')

# Проверим структуру данных и наличие необходимых столбцов
print("Исходная структура данных:")
print(data.info())

# Фильтруем только выданные заявки (issued == 1)
issued_apps = data[data['issued'] == 1]

# Анализируем распределение сроков рассмотрения заявок
tat_days = issued_apps['tat_days']

# Вычисляем медиану срока рассмотрения
median_tat = np.median(tat_days)
print(f"Медиана срока рассмотрения: {median_tat:.2f} дней")

# Вычисляем 90-й перцентиль срока рассмотрения
p90_tat = np.percentile(tat_days, 90)
print(f"Показатель 90-го перцентиля срока рассмотрения: {p90_tat:.2f} дней")

# Выводим одну строку для отчета
report_line = f"Медианное значение срока рассмотрения заявки: {median_tat:.2f}, 90-й перцентиль: {p90_tat:.2f}"
print(report_line)
```
