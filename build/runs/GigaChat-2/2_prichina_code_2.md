import pandas as pd
import numpy as np

# Загружаем данные из CSV-файла
df = pd.read_csv('data.csv')

# Проверяем структуру данных
print("Структура данных:")
print(df.head())
print("\nКоличество строк и столбцов:", df.shape)

# Устанавливаем index
df.set_index('week_start', inplace=True)

# Выбираем переменные для анализа
X1 = 'calls_th'  # обращения в контакт-центр
X2 = 'churn_pct'  # отток по дебетовым картам
Z = 'app_error_rate_pct'  # доля сессий с ошибкой в мобильном приложении

# Рассчитываем ковариацию между X1 и X2 при фиксированном Z
cov_XYZ = df[[X1, X2, Z]].cov()

# Рассчитываем дисперсию X и Y отдельно
sigma_X = df[X1].var()
sigma_Y = df[X2].var()

# Используем формулу частной корреляции Спирмена
rho_spirman = cov_XYZ.loc[X1, X2] / np.sqrt(cov_XYZ.loc[X1, X1] * cov_XYZ.loc[X2, X2])

# Выводим результаты расчета
print(f"\nЧастичная корреляция Спирмена между {X1} и {X2},"
      f"при фиксированном значении {Z}:")
print(rho_spirman)

# Формулируем строку для отчета
report_string = (f"Частичная корреляция Спирмена между обращениями в контакт-центр "
                 f"и оттоком по дебетовым картам, при фиксированной доле сессий "
                 f"с ошибкой в мобильном приложении составила {rho_spirman:.4f}.")

# Печать результатов и строки для отчёта
print(report_string)
