```python
import pandas as pd
import numpy as np

# Загружаем файл CSV в DataFrame df
df = pd.read_csv('data.csv')

# Проверяем структуру данных и типы переменных
print("Первая десятка записей:")
print(df.head(10))
print("\nТипы данных каждого столбца:")
print(df.dtypes)

# Фильтруем записи только с выданной заявкой (issued == 1)
filtered_df = df.query('issued == 1')

# Рассчитываем медиану срока рассмотрения заявок (tat_days)
median_tat = filtered_df['tat_days'].median()
print(f"Медиана срока рассмотрения заявок: {median_tat:.2f} дней")

# Рассчитываем 90-й перцентиль срока рассмотрения заявок
ninetieth_percentile = np.percentile(filtered_df['tat_days'], 90)
print(f"90-й перцентиль срока рассмотрения заявок: {ninetieth_percentile:.2f} дней")

# Формулировка результата для отчета
report_output = f"""
Медиана срока рассмотрения заявок: {median_tat:.2f} дней
90-й перцентиль срока рассмотрения заявок: {ninetieth_percentile:.2f} дней
"""

print(report_output)

````

Этот скрипт загружает данные из CSV-файла, фильтрует только выданные заявки и рассчитывает медианный срок рассмотрения заявок и 90-й перцентиль этого показателя. Выводится отчетный вывод, готовый для вставки в документ.
