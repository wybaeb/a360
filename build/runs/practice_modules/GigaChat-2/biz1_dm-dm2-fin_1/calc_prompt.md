```
Напиши файл расчёта rasschet.js для готового мини-инструмента «Параметры проекта: Портфель платёжных операций: восстановление объёма». Инструмент сам читает CSV-файлы и показывает результат; от тебя — одна функция расчёта входов.

Инструмент вызывает window.computeInputs(tables) и ждёт объект с числами: {volume: число, drop: число, fee: число}.
tables — объект: ключ — имя файла, значение — массив строк-объектов; ключи строки — названия столбцов из заголовка файла; числа уже переведены в числа (запятая заменена на точку), месяц — строка вида «2025-01»; строки идут в порядке файла.

Файл 1: tables["volume_dm.csv"] — вход volume («Объём операций в месяц», млн руб. в месяц). Первые строки файла:

2024-01;3452,2
2024-02;3365,4
2024-03;3324,3
Расчёт volume: среднее значение столбца объём_операций_млн_руб за последние 12 строк (месяцев).

Файл 2: tables["drop_dm2.csv"] — вход drop («Доля просадки к прошлому году», доля от 0 до 1). Первые строки файла:
﻿месяц;объём_операций_млн_руб;недобор_к_прошлому_году_млн_руб
2024-01;3452;281
2024-02;3365;285
2024-03;3324;275
Расчёт drop: сумма столбца недобор_к_прошлому_году_млн_руб ÷ сумма столбца объём_операций_млн_руб по всем строкам.

Файл 3: tables["fee_fin.csv"] — вход fee («Ставка комиссии», доля от объёма). Первые строки файла:
﻿параметр;значение
средняя_комиссия_проц;0,5
Расчёт fee: средняя_комиссия_проц ÷ 100.

Требования:
— только JavaScript, без HTML и без пояснений; объявить функцию именно так: window.computeInputs = function (tables) { ... };
— столбцы брать по названию из заголовка: row["закрытые_счета"]; имена переменных латиницей, комментарии по-русски;
— async и await, fetch и внешние библиотеки не использовать;
— если файла нет (tables["имя"] не определён) — вернуть для этого входа null, не бросать ошибку;
— «последние 12 строк» — это последние 12 элементов массива строк (arr.slice(-12)); при свёртке по месяцам сначала сложить значения строк с одинаковым месяцем, затем взять последние 12 месяцев по порядку.

Образец оформления (другой проект, другие файлы и столбцы):

  return { a: aVal, b: bVal, c: cVal };
};

# ответ

```
window.computeInputs = function (tables) {
  const f1 = tables['volume_dm.csv'];
  let volume = null;
  if (f1) {
    // Вычисляем объем операций за последние 12 месяцев
    const rows = f1.slice(-12);
    const sumVolume = rows.reduce((acc, r) => acc + Number(r['объём_операций_млн_руб'].replace(',', '.')), 0);
    volume = sumVolume / rows.length;
  }
  
  const f2 = tables['drop_dm2.csv'];
  let drop = null;
  if (f2) {
    // Расчет доли недоборной части относительно объема прошлого года
    const totalOper = f2.map(row => Number(row['объём_операций_млн_руб'].replace(',', '.')));
    const totalDrop = f2.map(row => Number(row['недобор_к_прошлому_году_млн_руб'].replace(',', '.')));
    const numerator = totalDrop.reduce((acc, v) => acc + v, 0);
    const denominator = totalOper.reduce((acc, v) => acc + v, 0);
    drop = denominator > 0 ? numerator / denominator : null;
  }

  const f3 = tables['fee_fin.csv'];
  let fee = null;
  if (f3) {
    // Ставка комиссии, деленная на 100
    const finTable = f3[0];
    const commissionRate = finTable['средняя_комиссия_проц'].replace(',', '.');
    fee = parseFloat(commissionRate) / 100;
  }

  return { volume, drop, fee };
}
