```python
import pandas as pd
import numpy as np

# Чтение исходного набора данных
data = pd.read_csv('your_data.csv')

# Вычисляем медиану и среднее абсолютное отклонение от медианы (MAD)
median_tat_min = data['tat_min'].median()
mad_tat_min = np.median(np.abs(data['tat_min'] - median_tat_min))

# Константа для масштабирования (примерно эквивалентна стандартному отклонению)
k = 1.4826  # Константа для нормального распределения

# Робастный аналог Z-оценки
robust_z_scores = k * ((data['tat_min'] - median_tat_min) / mad_tat_min)

# Определение порога аномалии (например, порог ±5 соответствует сильным выбросам)
threshold = 5

# Поиск значений с высоким робастным Z-score (выбросы)
outliers = data[np.abs(robust_z_scores) > threshold]

# Вывод количества найденных выбросов
print(f'Количество выявленных выбросов: {len(outliers)}')

# Итоговая фраза для отчёта
if len(outliers):
    print("Выявлены значительные отклонения сроков рассмотрения заявок.")
else:
    print("Значительных отклонений сроков рассмотрения заявок не выявлено.")
```
